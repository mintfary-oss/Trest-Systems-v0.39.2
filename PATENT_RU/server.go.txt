package api

import (
	"encoding/json"
	"net/http"
	"strings"

	"github.com/mintfary-oss/trest-sistems/internal/auth"

type Server struct {
	DB   *db.DB
	Auth *auth.Service
}

func NewServer(database *db.DB, a *auth.Service) *Server { return &Server{DB: database, Auth: a} }

func (s *Server) health(w http.ResponseWriter, _ *http.Request) {
	writeJSON(w, 200, map[string]string{"status": "ok"})
}
func (s *Server) ready(w http.ResponseWriter, r *http.Request) {
	if err := s.DB.PingContext(r.Context()); err != nil {
		writeJSON(w, 503, map[string]string{"status": "not_ready", "error": err.Error()})
		return
	}
	writeJSON(w, 200, map[string]string{"status": "ready"})
}

func (s *Server) users(w http.ResponseWriter, r *http.Request) {
	rows, err := s.DB.QueryContext(r.Context(), `SELECT id, email, name, role, created_at FROM users ORDER BY created_at DESC LIMIT 100`)
	if err != nil {
		writeError(w, err)
		return
	}
	defer rows.Close()
	out := make([]map[string]any, 0)
	for rows.Next() {
		var id, email, name, role string
		var created any
		if err := rows.Scan(&id, &email, &name, &role, &created); err != nil {
			writeError(w, err)
			return
		}
		out = append(out, map[string]any{"id": id, "email": email, "name": name, "role": role, "created_at": created})
	}
	if err := rows.Err(); err != nil {
		writeError(w, err)
		return
	}
	writeJSON(w, 200, out)
}

func (s *Server) projects(w http.ResponseWriter, r *http.Request) {
	rows, err := s.DB.QueryContext(r.Context(), `SELECT id, owner_id, name, status, created_at FROM projects ORDER BY created_at DESC LIMIT 100`)
	if err != nil {
		writeError(w, err)
		return
	}
	defer rows.Close()
	out := make([]map[string]any, 0)
	for rows.Next() {
		var id, owner, name, status string
		var created any
		if err := rows.Scan(&id, &owner, &name, &status, &created); err != nil {
			writeError(w, err)
			return
		}
		out = append(out, map[string]any{"id": id, "owner_id": owner, "name": name, "status": status, "created_at": created})
	}
	if err := rows.Err(); err != nil {
		writeError(w, err)
		return
	}
	writeJSON(w, 200, out)
}

func (s *Server) orders(w http.ResponseWriter, r *http.Request) {
	status := strings.TrimSpace(r.URL.Query().Get("status"))
	q := `SELECT id, project_id, object_id, title, status, amount, created_at FROM orders`
	args := []any{}
	if status != "" {
		q += ` WHERE status=$1`
		args = append(args, status)
	}
	q += ` ORDER BY created_at DESC LIMIT 100`
	rows, err := s.DB.QueryContext(r.Context(), q, args...)
	if err != nil {
		writeError(w, err)
		return
	}
	defer rows.Close()
	out := make([]map[string]any, 0)
	for rows.Next() {
		var id, project, title, status string
		var objectID any
		var amount any
		var created any
		if err := rows.Scan(&id, &project, &objectID, &title, &status, &amount, &created); err != nil {
			writeError(w, err)
			return
		}
		out = append(out, map[string]any{"id": id, "project_id": project, "object_id": objectID, "title": title, "status": status, "amount": amount, "created_at": created})
	}
	if err := rows.Err(); err != nil {
		writeError(w, err)
		return
	}
	writeJSON(w, 200, out)
}

func loggingJSON(next http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		w.Header().Set("X-Content-Type-Options", "nosniff")
		next.ServeHTTP(w, r)
	})
}
func writeJSON(w http.ResponseWriter, status int, v any) {
	w.Header().Set("Content-Type", "application/json")
	w.WriteHeader(status)
	_ = json.NewEncoder(w).Encode(v)
}
func writeError(w http.ResponseWriter, err error) {
	writeJSON(w, 500, map[string]string{"error": err.Error()})
}
