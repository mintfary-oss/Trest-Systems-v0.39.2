events {}
http {
  upstream trest_api { server api:8080; }
  upstream marketplace_api { server marketplace-api:8000; }
  upstream web { server web:3000; }
  server {
    listen 80;
    location /api/v1/ { proxy_pass http://trest_api; proxy_set_header Host $host; proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for; }
    location /marketplace-api/ { proxy_pass http://marketplace_api/; proxy_set_header Host $host; }
    location / { proxy_pass http://web; proxy_set_header Host $host; proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for; }
  }
}
