package queue

import (
	"context"
	"fmt"
	"os"
	"path/filepath"
	"strings"

	"github.com/jackc/pgx/v5"
	"github.com/jackc/pgx/v5/pgxpool"
	"github.com/mintfary-oss/trest-sistems/internal/bim/converter"
)

type Processor struct {
	DB      *pgxpool.Pool
	WorkDir string
}

func (p *Processor) RunOnce(ctx context.Context) (bool, error) {
	var id, op, format, input, output string
	tx, err := p.DB.Begin(ctx)
	if err != nil {
		return false, err
	}
	defer tx.Rollback(ctx)
	err = tx.QueryRow(ctx, `SELECT id,operation,format,input_uri,output_uri FROM bim_import_exports WHERE status='queued' ORDER BY created_at FOR UPDATE SKIP LOCKED LIMIT 1`).Scan(&id, &op, &format, &input, &output)
	if err == pgx.ErrNoRows {
		return false, nil
	}
	if err != nil {
		return false, err
	}
	if _, err = tx.Exec(ctx, `UPDATE bim_import_exports SET status='running' WHERE id=$1`, id); err != nil {
		return false, err
	}
	if err = tx.Commit(ctx); err != nil {
		return false, err
	}
	if err = p.process(ctx, op, format, input, output); err != nil {
		_, _ = p.DB.Exec(ctx, `UPDATE bim_import_exports SET status='failed',error=$2,completed_at=now() WHERE id=$1`, id, err.Error())
		return true, err
	}
	_, err = p.DB.Exec(ctx, `UPDATE bim_import_exports SET status='completed',completed_at=now(),error='' WHERE id=$1`, id)
	return true, err
}
func (p *Processor) process(ctx context.Context, op, format, input, output string) error {
	if op != "import" && op != "export" {
		return fmt.Errorf("unsupported operation %q", op)
	}
	if format == "gltf" || format == "glb" {
		return fmt.Errorf("native %s conversion adapter not enabled", format)
	}
	if format != "obj" {
		return fmt.Errorf("format %s adapter pending", format)
	}
	if op == "import" {
		return p.importOBJ(input, output)
	}
	return fmt.Errorf("OBJ export requires a BIM source adapter")
}
func (p *Processor) importOBJ(input, output string) error {
	if strings.TrimSpace(input) == "" {
		return fmt.Errorf("input_uri is empty")
	}
	r, err := os.Open(input)
	if err != nil {
		return err
	}
	defer r.Close()
	m, err := converter.ParseOBJ(r)
	if err != nil {
		return err
	}
	if output == "" {
		output = filepath.Join(p.WorkDir, filepath.Base(input)+".gltf")
	}
	if err = os.MkdirAll(filepath.Dir(output), 0755); err != nil {
		return err
	}
	w, err := os.Create(output)
	if err != nil {
		return err
	}
	defer w.Close()
	return converter.EncodeGLTF(w, m)
}
