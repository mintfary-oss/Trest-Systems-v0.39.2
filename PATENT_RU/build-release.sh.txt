#!/usr/bin/env bash
set -euo pipefail

ROOT="$(cd "$(dirname "$0")/.." && pwd)"
VERSION="${VERSION:-$(grep -o 'v0\.[0-9][0-9]*-stage[0-9][0-9]*' "$ROOT/CURRENT_STATE.md" | head -1 || echo 'dev')}"
OUT="$ROOT/release/bin"
mkdir -p "$OUT"

build_one() {
  local name="$1" pkg="$2" os="$3" arch="$4"
  mkdir -p "$OUT/$os-$arch"
  echo "building $name for $os/$arch"
  CGO_ENABLED=0 GOOS="$os" GOARCH="$arch" go build -trimpath -ldflags='-s -w' -o "$OUT/$os-$arch/$name" "$pkg"
}

# Run from a CI/release environment with dependencies already available.
build_one trest ./cmd/trest linux amd64
build_one trestctl ./cmd/trestctl linux amd64
build_one trest-api ./cmd/api linux amd64
build_one trest-worker ./cmd/worker linux amd64

( cd "$ROOT/release" && find bin -type f -exec sha256sum {} \; > checksums.sha256 )
printf '%s\n' "$VERSION" > "$ROOT/release/VERSION"
echo "release binaries prepared in $OUT"
